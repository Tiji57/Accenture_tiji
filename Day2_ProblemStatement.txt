Aggregation Pipeline Stages

// $match (Orders > 10000)
db.orders.aggregate([
  { $match: { totalAmount: { $gt: 10000 } } }
])

// $project (Customer & Amount only)
db.orders.aggregate([
  { $project: { customerName: 1, totalAmount: 1, _id: 0 } }
])

// $limit (First 5 results)
db.orders.aggregate([
  { $limit: 5 }
])

// $skip (First 3 orders)
db.orders.aggregate([
  { $skip: 3 }
])

// Sort (Total Amount in descending order)
db.orders.aggregate([
  { $sort: { totalAmount: -1 } }
])

// Group (Total Revenue by Category)
db.orders.aggregate([
  {
    $group: {
      _id: "$category",
      totalRevenue: { $sum: "$totalAmount" },
      avgAmount: { $avg: "$totalAmount" },
      maxAmount: { $max: "$totalAmount" },
      minAmount: { $min: "$totalAmount" },
      count: { $sum: 1 }
    }
  }
])

  { $match: { totalAmount: { $gt: 10000 } } },
  { $group: { _id: "$category", totalRevenue: { $sum: "$totalAmount" } } },
  { $project: { category: "$_id", totalRevenue: 1, _id: 0 } },
  { $sort: { totalRevenue: -1 } }
])
